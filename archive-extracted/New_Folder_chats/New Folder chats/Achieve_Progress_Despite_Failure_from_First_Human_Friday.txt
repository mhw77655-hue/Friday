name: construct

# One real construction task per run, in four separate permission zones.
#   worker    the only job that sees the model key. Read-only token. Produces a patch, nothing else.
#   candidate applies the patch, pushes a candidate branch, starts the real Android CI on it. No model key.
#   verify    runs the story's own acceptance command (from .ralph/prd.json) against that CI run. Read-only. No model key.
#   promote   only after verify passes: marks the story, opens a pull request. Never touches main.
# Nothing here retries by itself and nothing here waits for a model to come back.

on:
  workflow_dispatch:
    inputs:
      story:
        description: "Story id from .ralph/prd.json. Empty = the open story with the lowest priority number."
        required: false
        default: ""
      note:
        description: "Optional extra instruction for the worker, for example the failure evidence of the last run."
        required: false
        default: ""
      publish:
        description: "false = stop after the worker and only keep its patch (a cheap first look). true = the full run."
        type: boolean
        default: true

concurrency:
  group: construct
  cancel-in-progress: false

permissions:
  contents: read

jobs:
  worker:
    runs-on: ubuntu-latest
    timeout-minutes: 75
    outputs:
      story: ${{ steps.pick.outputs.story }}
      title: ${{ steps.pick.outputs.title }}
      red: ${{ steps.pick.outputs.red }}
      base: ${{ steps.pick.outputs.base }}
    steps:
      - name: Checkout
        uses: actions/checkout@v4
        with:
          fetch-depth: 0
          persist-credentials: false

      - name: Pick the story and build the work base
        id: pick
        env:
          IN_STORY: ${{ inputs.story }}
        run: |
          set -euo pipefail
          ID="$IN_STORY"
          if [ -z "$ID" ]; then
            ID=$(jq -r '[.userStories[] | select(.passes != true)] | sort_by(.priority) | .[0].id // empty' .ralph/prd.json)
          fi
          [ -n "$ID" ] || { echo "No open story in .ralph/prd.json"; exit 1; }
          jq -e --arg id "$ID" '.userStories[] | select(.id == $id)' .ralph/prd.json > /dev/null || { echo "Unknown story $ID"; exit 1; }
          [ "$(jq -r --arg id "$ID" '.userStories[] | select(.id == $id) | .passes' .ralph/prd.json)" != "true" ] || { echo "Story $ID is already closed"; exit 1; }
          TITLE=$(jq -r --arg id "$ID" '.userStories[] | select(.id == $id) | .title' .ralph/prd.json)
          RED=$(jq -r --arg id "$ID" '.userStories[] | select(.id == $id) | .verifyCommand' .ralph/prd.json | grep -oE 'refs/heads/audit/[A-Za-z0-9._/-]+' | sed 's#^refs/heads/##' | sort -u | head -1 || true)
          BASE=$(git rev-parse HEAD)
          git config user.name "friday-construct"
          git config user.email "friday-construct@users.noreply.github.com"
          if [ -n "$RED" ]; then
            git fetch --no-tags origin "+refs/heads/$RED:refs/remotes/origin/$RED"
            git merge --no-ff -m "construct: red tests ($RED) under the green work" "origin/$RED"
          fi
          {
            echo "story=$ID"
            echo "title=$TITLE"
            echo "red=$RED"
            echo "base=$BASE"
          } >> "$GITHUB_OUTPUT"
          jq --arg id "$ID" '.userStories[] | select(.id == $id) | del(.notes)' .ralph/prd.json > "$RUNNER_TEMP/story.json"
          echo "Story $ID: $TITLE (red branch: ${RED:-none})" >> "$GITHUB_STEP_SUMMARY"

      - name: Write the construction contract
        env:
          IN_NOTE: ${{ inputs.note }}
          RED: ${{ steps.pick.outputs.red }}
        run: |
          set -euo pipefail
          mkdir -p .construct
          echo ".construct/" >> .git/info/exclude
          cat > .construct/prompt.md <<'EOF'
          You are the construction worker for the Friday project. Read AGENTS.md first, then .ralph/VISION.md.
          Build exactly ONE story in this checkout: the story below, nothing else.

          How this run is organised (this differs from .ralph/prompt.md, which was written for the old phone loop):
          - You only edit files in the working tree. Do NOT run git commit, git push or git checkout, and do not create branches.
          - You have no network and must not run Gradle. CI builds and tests the result after you finish (see AGENTS.md).
          - Do NOT edit .github/**, .ralph/prd.json, .ralph/ci_verify.sh, .ralph/ralph.sh or .ralph/cloud/**. The run fails if you do.
          - Do not mark the story as passing. A separate job does that after CI and the story's own acceptance command pass.
          - From .ralph/prompt.md use only the discipline sections: plan first; grep the real sources before every import and
            record fixes in .ralph/SYMBOL_FACTS.md; systematic debugging. Ignore its steps about committing, branches, prd.json and progress.txt.
          - If the story has a RED-first criterion and a red branch is named below, that criterion is already done: its test files are in
            the tree and must stay byte-identical. Build the remaining criteria only.
          - Create the files the acceptance criteria name, at the paths they name. Put the mechanisms-considered notes in the incident file
            under a heading "Mechanisms considered".

          Finish with a final message of at most 30 lines: the files you changed, how each acceptance criterion is met or why it is not,
          and what remains unproven. Do not claim anything works: CI decides.
          EOF
          {
            echo
            echo "## Story (from .ralph/prd.json)"
            cat "$RUNNER_TEMP/story.json"
            echo
            echo "## Red branch already merged under this work: ${RED:-none}"
            if [ -n "$IN_NOTE" ]; then
              echo
              echo "## Extra instruction from the operator"
              printf '%s\n' "$IN_NOTE"
            fi
          } >> .construct/prompt.md

      - name: Run Codex
        id: codex
        uses: openai/codex-action@v1
        timeout-minutes: 50
        with:
          openai-api-key: ${{ secrets.OPENAI_API_KEY }}
          permission-profile: ":workspace"
          prompt-file: .construct/prompt.md
          output-file: .construct/final-message.md
          model: ${{ vars.CODEX_MODEL }}
          effort: ${{ vars.CODEX_EFFORT }}

      - name: Capture the candidate patch
        run: |
          set -euo pipefail
          git add -A
          mkdir -p cand
          git diff --cached --binary > cand/candidate.patch
          [ -s cand/candidate.patch ] || { echo "Codex changed nothing, so there is nothing to verify"; exit 1; }
          cp .construct/final-message.md cand/final-message.md 2>/dev/null || echo "(no final message)" > cand/final-message.md
          {
            echo "### Worker finished"
            git diff --cached --stat | tail -40
          } >> "$GITHUB_STEP_SUMMARY"

      - name: Keep the patch
        uses: actions/upload-artifact@v4
        with:
          name: candidate
          path: cand/
          if-no-files-found: error

  candidate:
    needs: worker
    if: ${{ inputs.publish }}
    runs-on: ubuntu-latest
    timeout-minutes: 20
    permissions:
      contents: write
      actions: write
    outputs:
      branch: ${{ steps.push.outputs.branch }}
    steps:
      - name: Checkout the exact base the worker used
        uses: actions/checkout@v4
        with:
          ref: ${{ needs.worker.outputs.base }}
          fetch-depth: 0

      - name: Fetch the patch
        uses: actions/download-artifact@v4
        with:
          name: candidate
          path: cand

      - name: Build the candidate branch, push it and start the real CI on it
        id: push
        env:
          GH_TOKEN: ${{ github.token }}
          STORY: ${{ needs.worker.outputs.story }}
          RED: ${{ needs.worker.outputs.red }}
        run: |
          set -euo pipefail
          git config user.name "friday-construct"
          git config user.email "friday-construct@users.noreply.github.com"
          BR="candidate/${STORY}-${GITHUB_RUN_ID}"
          git checkout -q -b "$BR"
          if [ -n "$RED" ]; then
            git fetch --no-tags origin "+refs/heads/$RED:refs/remotes/origin/$RED"
            git merge --no-ff -m "construct: red tests ($RED) under the green work" "origin/$RED"
