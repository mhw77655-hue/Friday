The technology to use: MCP
MCP is specifically designed to let AI applications connect to common external tools and data sources. Anthropic describes it as a standard interface between AI applications and data/tools. 
There are already Obsidian MCP implementations that expose an Obsidian vault to both Claude and ChatGPT, including search, reading and writing. 
One particularly interesting project is Vault MCP Connector, which is specifically designed around connecting Obsidian to ChatGPT, Claude, Codex and other MCP clients. 
Claude side
Claude is actually straightforward here.
Anthropic currently supports remote MCP/custom connectors on Claude and Claude Desktop for Pro, Max, Team and Enterprise, and Claude can connect directly to a remote MCP server. 
So Claude → Obsidian can be:
Claude
   ↓
Remote MCP
   ↓
JARVIS Obsidian MCP
   ↓
Obsidian vault

ChatGPT side — there's one catch
OpenAI currently supports custom MCP apps/connectors in ChatGPT, but full MCP with write/modify actions is currently rolling out for Business, Enterprise and Edu, rather than being universally available on consumer plans. OpenAI also states that ChatGPT connects to remote MCP servers rather than directly to a private/local MCP server. 
So on your current ChatGPT Free setup, I would not build the architecture around expecting the ChatGPT mobile app to directly mount your local Obsidian vault.
Instead, there are two layers:
                 ┌───────────────┐
                 │   Obsidian    │
                 └───────┬───────┘

              ┌──────────┴──────────┐

                                  │
                               ChatGPT

OpenAI's API supports remote MCP tools, so the API route can give an OpenAI model access to the exact same memory layer. 
That is actually better for your JARVIS architecture than trying to force the consumer ChatGPT application to become the central brain.

from fastapi import FastAPI, HTTPException
from pydantic import BaseModel
from pathlib import Path
from typing import Optional
from datetime import datetime, timezone
import re
import urllib.parse

app = FastAPI(
    title="JARVIS Memory Server",
    version="1.0.0"
)

VAULT = Path.home() / "storage" / "shared" / "Documents" / "JARVIS"


class MemoryWrite(BaseModel):
    path: str
    content: str
    source: str = "jarvis"
    memory_type: str = "memory"


class MemoryUpdate(BaseModel):
    content: str


class MemorySearch(BaseModel):
    query: str
    folder: Optional[str] = None
    limit: int = 20


def safe_path(relative_path: str) -> Path:
    relative_path = relative_path.replace("\\", "/").lstrip("/")

    path = (VAULT / relative_path).resolve()

    if not str(path).startswith(str(VAULT.resolve())):
        raise HTTPException(
            status_code=400,
            detail="Invalid path"
        )

    return path


def metadata(source: str, memory_type: str) -> str:
    now = datetime.now(timezone.utc).isoformat()

    return (
        "---\n"
        f"created: {now}\n"
        f"updated: {now}\n"
        f"source: {source}\n"
        f"type: {memory_type}\n"
        "---\n\n"
    )


@app.get("/")
def root():
    return {
        "service": "JARVIS Memory Server",
        "status": "online",
        "vault": str(VAULT)
    }


@app.get("/health")
def health():
    return {
        "status": "ok",
        "vault_exists": VAULT.exists()
    }


@app.get("/files")
def files(folder: str = ""):
    base = safe_path(folder)

    if not base.exists():
        raise HTTPException(
            status_code=404,
            detail="Folder not found"
        )

    results = []

    for p in base.rglob("*"):
        if p.is_file() and p.suffix.lower() == ".md":
            results.append(
                str(p.relative_to(VAULT))
            )

    return {
        "count": len(results),
        "files": results
    }


@app.get("/memory/read")
def read_memory(path: str):
    file_path = safe_path(path)

    if not file_path.exists():
        raise HTTPException(
            status_code=404,
            detail="Memory not found"
        )

    if not file_path.is_file():
        raise HTTPException(
            status_code=400,
            detail="Not a file"
        )

    return {
        "path": str(file_path.relative_to(VAULT)),
        "content": file_path.read_text(
            encoding="utf-8"
        )
    }


@app.post("/memory/write")
def write_memory(memory: MemoryWrite):
    file_path = safe_path(memory.path)

    if file_path.suffix.lower() != ".md":
        file_path = file_path.with_suffix(".md")

    file_path.parent.mkdir(
        parents=True,
        exist_ok=True
    )

    content = (
        metadata(
            memory.source,
            memory.memory_type
        )
        + memory.content
    )

    file_path.write_text(
        content,
        encoding="utf-8"
    )

    return {
        "status": "written",
        "path": str(file_path.relative_to(VAULT))
    }


@app.put("/memory/update")
def update_memory(
    path: str,
    memory: MemoryUpdate
):
    file_path = safe_path(path)

    if not file_path.exists():
        raise HTTPException(
            status_code=404,
            detail="Memory not found"
        )

    file_path.write_text(
        memory.content,
        encoding="utf-8"
    )

    return {
        "status": "updated",
        "path": str(file_path.relative_to(VAULT))
    }


@app.post("/memory/append")
def append_memory(
    path: str,
    content: str
):
    file_path = safe_path(path)

    if not file_path.exists():
        raise HTTPException(
            status_code=404,
            detail="Memory not found"
        )

    with file_path.open(
        "a",
        encoding="utf-8"
    ) as f:
        f.write("\n\n" + content)

    return {
        "status": "appended",
        "path": str(file_path.relative_to(VAULT))
    }


@app.delete("/memory/delete")
def delete_memory(path: str):
    file_path = safe_path(path)

    if not file_path.exists():
        raise HTTPException(
            status_code=404,
            detail="Memory not found"
        )

    file_path.unlink()

    return {
        "status": "deleted",
        "path": str(file_path.relative_to(VAULT))
    }


@app.post("/memory/search")
def search_memory(search: MemorySearch):
    query = search.query.lower().strip()

    if not query:
        return {
            "results": []
        }

    base = VAULT

    if search.folder:
        base = safe_path(search.folder)

    results = []

    for file_path in base.rglob("*.md"):
        try:
            content = file_path.read_text(
                encoding="utf-8"
            )
        except Exception:
            continue

        lower = content.lower()

        if query in lower:
            lines = content.splitlines()

            matches = []

            for i, line in enumerate(lines):
                if query in line.lower():
                    start = max(0, i - 2)
                    end = min(
                        len(lines),
                        i + 3
                    )

                    matches.append(
                        "\n".join(
                            lines[start:end]
                        )
                    )

            results.append({
                "path": str(
                    file_path.relative_to(VAULT)
                ),
                "matches": matches[:5]
            })

        if len(results) >= search.limit:
            break

    return {
        "query": search.query,
        "count": len(results),
        "results": results
    }


@app.get("/memory/recent")
def recent(limit: int = 20):
    files = [
        p for p in VAULT.rglob("*.md")
        if p.is_file()
    ]

    files.sort(
        key=lambda p: p.stat().st_mtime,
        reverse=True
    )

    results = []

    for p in files[:limit]:
        results.append({
            "path": str(
                p.relative_to(VAULT)
            ),
            "modified": datetime.fromtimestamp(
                p.stat().st_mtime,
                tz=timezone.utc
            ).isoformat()
        })

    return {
        "results": results
    }
