def differences(base):
    bm = {s['id']: s for s in base['userStories']}
    problems, undone = [], []
    for w in work['userStories']:
        f = bm.get(w['id'])
        if f is None:
            problems.append('story only in the working copy: ' + w['id'])
        elif w != f:
            diff = sorted(k for k in set(w) | set(f) if w.get(k) != f.get(k))
            if set(diff) <= {'passes', 'notes'} and 'AUTO-REJECTED' in (w.get('notes') or ''):
                undone.append(w['id'])
            else:
                problems.append(w['id'] + ' differs in ' + ', '.join(diff))
    if {k: v for k, v in work.items() if k != 'userStories'} != {k: v for k, v in base.items() if k != 'userStories'}:
        problems.append('top-level fields differ')
    return problems, undone
