cd /sdcard/jarvis-repo
OUT=".ralph/REPO_INVENTORY_$(date +%Y%m%d_%H%M%S).md"
{
  echo "# JARVIS Repo Inventory - $(date)"
  echo ""
  echo "## Summary"
  echo "Total git-tracked files: $(git ls-files | wc -l)"
  echo ""
  echo "## Per-file inventory"
  echo ""
  git ls-files | while read -r f; do
    [ -f "$f" ] || continue
    if file "$f" | grep -qi text; then
      lines=$(wc -l < "$f" 2>/dev/null | tr -d ' ')
      size=$(du -h "$f" 2>/dev/null | cut -f1)
      echo "### $f"
      echo "- lines: $lines | size: $size"
      case "$f" in
        *.kt|*.java)
          decls=$(grep -nE "^\s*(class|interface|object|fun|enum class|sealed class|data class|sealed interface)\s" "$f")
          [ -n "$decls" ] && { echo "- declarations:"; echo "$decls" | sed 's/^/    /'; }
          ;;
        *.py)
          decls=$(grep -nE "^\s*(class|def)\s" "$f")
          [ -n "$decls" ] && { echo "- declarations:"; echo "$decls" | sed 's/^/    /'; }
          ;;
      esac
      echo ""
    else
      size=$(du -h "$f" 2>/dev/null | cut -f1)
      echo "### $f (binary, $size)"
      echo ""
    fi
  done
} > "$OUT"
cp "$OUT" /sdcard/Download/
