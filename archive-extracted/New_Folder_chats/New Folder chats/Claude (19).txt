import urllib.request, json, os

def download_repo(repo_id, local_dir, skip_prefixes=()):
    api_url = f"https://huggingface.co/api/models/{repo_id}"
    with urllib.request.urlopen(api_url) as r:
        data = json.loads(r.read())
    for f in data["siblings"]:
        path = f["rfilename"]
        if any(path.startswith(p) for p in skip_prefixes):
            continue
        local_path = os.path.join(local_dir, path)
        os.makedirs(os.path.dirname(local_path) or ".", exist_ok=True)
        if os.path.exists(local_path):
            print(f"skip (exists): {path}")
            continue
        url = f"https://huggingface.co/{repo_id}/resolve/main/{path}"
        print(f"downloading: {path}")
        urllib.request.urlretrieve(url, local_path)
    print(f"done: {repo_id}")

download_repo("myshell-ai/OpenVoice", "./OpenVoice",
              skip_prefixes=("checkpoints/base_speakers/ZH",))
download_repo("Rabe3/kemetone", "./kemetone",

import urllib.request, urllib.parse, json, os

def download_repo(repo_id, local_dir, skip_prefixes=()):
    api_url = f"https://huggingface.co/api/models/{repo_id}"
    with urllib.request.urlopen(api_url) as r:
        data = json.loads(r.read())
    for f in data["siblings"]:
        path = f["rfilename"]
        if any(path.startswith(p) for p in skip_prefixes):
            continue
        local_path = os.path.join(local_dir, path)
        os.makedirs(os.path.dirname(local_path) or ".", exist_ok=True)
        if os.path.exists(local_path):
            print(f"skip (exists): {path}")
            continue
        encoded_path = urllib.parse.quote(path)
        url = f"https://huggingface.co/{repo_id}/resolve/main/{encoded_path}"
        print(f"downloading: {path}")
        urllib.request.urlretrieve(url, local_path)
    print(f"done: {repo_id}")

download_repo("myshell-ai/OpenVoice", "./OpenVoice",
              skip_prefixes=("checkpoints/base_speakers/ZH",))
download_repo("Rabe3/kemetone", "./kemetone",
              skip_prefixes=("assets/", "samples/"))
